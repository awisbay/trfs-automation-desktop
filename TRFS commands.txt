
******************************
*******FOR NR700 *************


*********** VSWR *************
sdir
get B28_RRU.*,rfport=[A-D] vswrSupervisionSensitivity        
get B28_RRU.*,rfport=[A-D],freq* vswrSupervisionSensitivity

*********** CELL *************
st nrcelldu=.*P-
st B28
(ENM CELL M SS)
*********** NOC LOGS *********
get N28.* bschannelBW	
hget Carrier=N28 noOfUsedTxAntennas|noOfUsedRxAntennas
lpr GUtranCellRelation=5152-0000000000902002-50
cvcu
(ENM SHM SW SS)

************ ALARM **********
alt
(ENM FM SS)


************* PIM ***********
pmxhetd NRCellDU=TCPHTP3ACANOCOTAGUMDDNP  Int_AvgRadioRecInterferencePower$

************* RET ***********
hgetc RetSubUnit=.* electricalAntennaTilt|operationalState|userLabel

*******FOR G900 *************

st sec
hget . gsmsectorId|bscNodeIdentity|gsmsectorid
(ENM CELL M SS)

*********** NOC LOGS *********

cvcu
(ENM SHM SW SS)


************ ALARM **********
alt
(ENM FM SS)

rxtcp:mo=rxstg-310;
rxtcp:mo=rxstg-311;
rxtcp:mo=rxstg-312;
rlcrp:cell=M31179S1;
rlcrp:cell=M31179S2;
rlcrp:cell=M31179S3;

************* RET ***********
hgetc RetSubUnit=.* electricalAntennaTilt|operationalState|userLabel
